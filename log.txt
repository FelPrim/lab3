BUILD.sh
#!/usr/bin/env bash
set -o pipefail
MAKE_CMD="mingw32-make"
MAKE_FLAGS="-j1 V=1"
MAX_RETRIES=8
BUILD_LOG="build.log"

export SHELL=/usr/bin/bash
export MSYS2_ARG_CONV_EXCL="*"

attempt=1
while [ $attempt -le $MAX_RETRIES ]; do
  rm -f "$BUILD_LOG"
  if ! $MAKE_CMD $MAKE_FLAGS 2>&1 | tee "$BUILD_LOG"; then
    echo "make failed — анализируем лог..."
  else
    echo "make OK"; exit 0
  fi

  missing_files=$(grep -oE 'ar: @[A-Za-z0-9_./-]+\.a\.objs' "$BUILD_LOG" || true)
  [ -n "$missing_files" ] || { echo "Нет ar:@*.a.objs ошибок — см. $BUILD_LOG"; exit 1; }

  echo "$missing_files" | sed 's/^ar: @//' | sort -u | while read -r entry; do
    entry=${entry#@}
    libdir=$(dirname "$entry")
    libfile=$(basename "$entry")
    archive="${libfile%.a.objs}.a"

    [ -d "$libdir" ] || { echo "Каталог $libdir не найден, пропускаю"; continue; }

    echo "Fixing $libdir/$archive ..."

    # переход в каталог и сборка рекурсивного списка .o (относительные пути внутри каталога)
    pushd "$libdir" > /dev/null
    # find выдаёт ./sub/obj.o — удаляем ведущий ./ чтобы ar видел относительные пути без лишних префиксов
    find . -type f -name '*.o' -print | sed 's|^\./||' > "$libfile"
    cnt=$(wc -l < "$libfile" | tr -d ' ')
    if [ "$cnt" -eq 0 ]; then
      echo "В $libdir нет .o — пропускаем"
      popd > /dev/null
      continue
    fi

    /mingw64/bin/ar rc "$archive" @"$libfile" && /mingw64/bin/ranlib "$archive"
    popd > /dev/null
  done

  attempt=$((attempt+1))
done

echo "Превышено количество попыток ($MAX_RETRIES). Смотри $BUILD_LOG."
exit 1

make install

#!/usr/bin/env bash
# ffbuild-win-wrap.sh
# Usage: ./ffbuild-win-wrap.sh [make args...]
# Example: ./ffbuild-win-wrap.sh -j1 V=1

set -euo pipefail

# Thresholds (подправьте при необходимости)
CMD_LEN_LIMIT=8000        # если суммарная длина аргументов > этого — триггерим разбиение
OBJ_COUNT_LIMIT=200       # если объектов больше — тоже триггерим разбиение

WORKDIR="$(pwd)"
WRAPDIR="$(mktemp -d "${WORKDIR}/ffwrap.XXXX")"
trap 'rm -rf -- "$WRAPDIR"' EXIT

# helper: find the real binary that is not our wrapper
find_real_binary() {
  name="$1"
  # search PATH entries and return first executable $name that is not in $WRAPDIR
  OLDIFS="$IFS"; IFS=':'; 
  for p in $PATH; do
    candidate="$p/$name"
    if [ -x "$candidate" ] && [ "$candidate" != "$WRAPDIR/$name" ]; then
      echo "$candidate"
      IFS="$OLDIFS"
      return 0
    fi
  done
  IFS="$OLDIFS"
  # fallback to `command -v`
  command -v "$name" || return 1
}

REAL_AR="$(find_real_binary ar)"
REAL_GCC="$(find_real_binary gcc)"

if [ -z "$REAL_AR" ] || [ -z "$REAL_GCC" ]; then
  echo "ERROR: cannot find real 'ar' or 'gcc' in PATH" >&2
  exit 1
fi

# -------------------------
# AR wrapper
# -------------------------
cat >"$WRAPDIR/ar" <<'AR_WRAPPER'
#!/usr/bin/env bash
# ar wrapper: if many input files / long argv, add them to archive in chunks

set -euo pipefail

SELF="$(readlink -f "$0")"
# find real ar by searching PATH for first 'ar' not equal to this
for p in $(echo "$PATH" | tr ':' '\n'); do
  if [ -x "$p/ar" ] && [ "$(readlink -f "$p/ar")" != "$SELF" ]; then
    REAL_AR="$p/ar"
    break
  fi
done
: "${REAL_AR:?real ar not found}"

# Minimal parsing: command is like: ar [operation] archive-file [members...]
# We'll forward everything unchanged unless members list is long.
op="$1"
archive="$2"
shift 2
members=( "$@" )

# compute combined length
combined_len=0
for m in "${members[@]}"; do combined_len=$((combined_len + ${#m})); done

MAXLEN=7000
MAXCOUNT=200
CHUNK=150

if [ "${#members[@]}" -eq 0 ] || ( [ "$combined_len" -lt "$MAXLEN" ] && [ "${#members[@]}" -lt "$MAXCOUNT" ] ); then
  exec "$REAL_AR" "$op" "$archive" "${members[@]}"
fi

# otherwise: perform chunked ar
# start with first file to create/replace archive (use 'rcs' semantics: create)
# if op includes 's' or we need symbol index, we'll run 'ranlib' at the end if available
first=true
i=0
n=${#members[@]}
while [ $i -lt $n ]; do
  j=$((i+CHUNK))
  if [ $j -gt $n ]; then j=$n; fi
  chunk=( "${members[@]:i:CHUNK}" )
  if $first; then
    # create archive (use 'rcs' if available)
    "$REAL_AR" rcs "$archive" "${chunk[@]}"
    first=false
  else
    "$REAL_AR" r "$archive" "${chunk[@]}"
  fi
  i=$j
done

# try to run ranlib if exists and if op requested index
if command -v ranlib >/dev/null 2>&1; then
  command -v ranlib >/dev/null 2>&1 && ranlib "$archive" || true
fi

exit 0

#!/usr/bin/env bash
# gcc wrapper: if final link invocation has too many object files, make a linker response file
set -euo pipefail

SELF="$(readlink -f "$0")"
for p in $(echo "$PATH" | tr ':' '\n'); do
  if [ -x "$p/gcc" ] && [ "$(readlink -f "$p/gcc")" != "$SELF" ]; then
    REAL_GCC="$p/gcc"
    break
  fi
done
: "${REAL_GCC:?real gcc not found}"

args=( "$@" )

# quick pass: detect if this call is a link (has -o and at least one .o and no -c)
is_link=false
has_o=false
has_c=false
obj_count=0
total_len=0
for a in "${args[@]}"; do
  case "$a" in
    -c) has_c=true;;
    -o) has_o=true;; # note: -o might be followed by filename; we don't need exact parsing here
  esac
  if [[ "$a" == *.o ]]; then
    obj_count=$((obj_count+1))
    total_len=$((total_len + ${#a}))
  fi
done
if ! $has_c && $has_o && [ $obj_count -gt 0 ]; then
  is_link=true
fi

# thresholds (tweak if necessary)
MAXLEN=7000
MAXCOUNT=200

if ! $is_link || ( [ $obj_count -lt $MAXCOUNT ] && [ $total_len -lt $MAXLEN ] ); then
  exec "$REAL_GCC" "${args[@]}"
fi

# We need to build a response file for the linker with the list of object files.
# Strategy: collect all tokens that look like existing files ending with .o (best-effort),
# replace that contiguous set with -Wl,@/tmp/respfile
RSP="$(mktemp /tmp/gcc-link-args.XXXX)"
# Put each object on a new line
for a in "${args[@]}"; do
  if [[ "$a" == *.o && -f "$a" ]]; then
    echo "$a" >> "$RSP"
  fi
done

# Build new args: replace the object filenames by single token -Wl,@$RSP
newargs=()
obj_inserted=false
for a in "${args[@]}"; do
  if [[ "$a" == *.o && -f "$a" ]]; then
    if ! $obj_inserted; then
      newargs+=( "-Wl,@$RSP" )
      obj_inserted=true
    fi
    # skip this obj (we already added them via rsp)
  else
    newargs+=( "$a" )
  fi
done

# Call real gcc with modified args
exec "$REAL_GCC" "${newargs[@]}"
